  <script>
    // State management
    let state = {
      accounts: {
        bbva: 3500.00,
        myinvestor: 18200.00,
        trade_republic: 8500.00
      },
      transactions: []
    };

    // Load from localStorage on start
    function init() {
      const saved = localStorage.getItem('mi_patrimonio_data');
      if (saved) {
        try {
          state = JSON.parse(saved);
        } catch (e) {
          console.error("Error cargando localStorage", e);
        }
      }
      render();
    }

    // Save state to localStorage
    function save() {
      localStorage.setItem('mi_patrimonio_data', JSON.stringify(state));
      render();
    }

    // Render UI
    function render() {
      const total = state.accounts.bbva + state.accounts.myinvestor + state.accounts.trade_republic;
      
      document.getElementById('totalBalance').innerText = total.toLocaleString('es-ES', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) + ' €';
      document.getElementById('bbvaBalance').innerText = state.accounts.bbva.toLocaleString('es-ES', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) + ' €';
      document.getElementById('myinvestorBalance').innerText = state.accounts.myinvestor.toLocaleString('es-ES', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) + ' €';
      document.getElementById('trBalance').innerText = state.accounts.trade_republic.toLocaleString('es-ES', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) + ' €';

      document.getElementById('inputBBVA').value = state.accounts.bbva;
      document.getElementById('inputMyInvestor').value = state.accounts.myinvestor;
      document.getElementById('inputTR').value = state.accounts.trade_republic;

      updateCharts();
    }

    // Add Expense
    function addExpense(e) {
      e.preventDefault();
      const concept = document.getElementById('expenseConcept').value;
      const amount = parseFloat(document.getElementById('expenseAmount').value);
      const account = document.getElementById('expenseAccount').value;

      if (!amount || amount <= 0) return;

      state.accounts[account] -= amount;
      state.transactions.push({
        id: Date.now(),
        date: new Date().toISOString(),
        concept,
        amount,
        account
      });

      document.getElementById('expenseConcept').value = '';
      document.getElementById('expenseAmount').value = '';

      save();
    }

    // Update Account Balance directly
    function updateAccountBalance(acc) {
      let val = 0;
      if (acc === 'bbva') val = parseFloat(document.getElementById('inputBBVA').value);
      if (acc === 'myinvestor') val = parseFloat(document.getElementById('inputMyInvestor').value);
      if (acc === 'trade_republic') val = parseFloat(document.getElementById('inputTR').value);

      if (!isNaN(val)) {
        state.accounts[acc] = val;
        save();
      }
    }

    // Backup: Export JSON
    function exportData() {
      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(state, null, 2));
      const downloadAnchor = document.createElement('a');
      downloadAnchor.setAttribute("href", dataStr);
      downloadAnchor.setAttribute("download", `mi-patrimonio-backup-${new Date().toISOString().slice(0,10)}.json`);
      document.body.appendChild(downloadAnchor);
      downloadAnchor.click();
      downloadAnchor.remove();
    }

    // Backup: Import JSON
    function importData(event) {
      const fileReader = new FileReader();
      fileReader.onload = function(e) {
        try {
          const importedState = JSON.parse(e.target.result);
          if (importedState && importedState.accounts) {
            state = importedState;
            save();
            alert('¡Copia de seguridad importada con éxito!');
          } else {
            alert('El archivo seleccionado no tiene el formato correcto.');
          }
        } catch (err) {
          alert('Error al leer el archivo JSON.');
        }
      };
      if (event.target.files[0]) {
        fileReader.readAsText(event.target.files[0]);
      }
    }

    // Charts
    let chartDist = null;
    let chartEvol = null;

    function updateCharts() {
      // Distribution Donut Chart
      const ctxDist = document.getElementById('distributionChart').getContext('2d');
      if (chartDist) chartDist.destroy();
      chartDist = new Chart(ctxDist, {
        type: 'doughnut',
        data: {
          labels: ['BBVA', 'MyInvestor', 'Trade Republic'],
          datasets: [{
            data: [state.accounts.bbva, state.accounts.myinvestor, state.accounts.trade_republic],
            backgroundColor: ['#3b82f6', '#6366f1', '#10b981']
          }]
        },
        options: { responsive: true, maintainAspectRatio: false }
      });

      // Evolution Line Chart (Simulated layout)
      const ctxEvol = document.getElementById('evolutionChart').getContext('2d');
      if (chartEvol) chartEvol.destroy();
      const total = state.accounts.bbva + state.accounts.myinvestor + state.accounts.trade_republic;
      chartEvol = new Chart(ctxEvol, {
        type: 'line',
        data: {
          labels: ['Hace 3 Meses', 'Hace 2 Meses', 'Mes Pasado', 'Actual'],
          datasets: [{
            label: 'Patrimonio (€)',
            data: [total * 0.92, total * 0.95, total * 0.98, total],
            borderColor: '#4f46e5',
            tension: 0.3,
            fill: true,
            backgroundColor: 'rgba(79, 70, 229, 0.05)'
          }]
        },
        options: { responsive: true, maintainAspectRatio: false }
      });
    }

    // Start App
    init();
  </script>
